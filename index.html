function normalizeClientId(value) {
  var id = String(value || 'global').replace(/[^a-zA-Z0-9_\-]/g, '').substring(0, 80);
  return id || 'global';
}

function getAdminLoginCacheKeys(clientId) {
  var id = normalizeClientId(clientId);
  return { attempts: 'nexus_admin_attempts_' + id, locked: 'nexus_admin_locked_' + id };
}

function getAdminLockState(clientId) {
  var cache = CacheService.getScriptCache();
  var keys = getAdminLoginCacheKeys(clientId);
  var lockedUntil = parseInt(cache.get(keys.locked) || '0', 10);
  var now = new Date().getTime();
  if (lockedUntil && now < lockedUntil) {
    return { locked: true, remaining_seconds: Math.ceil((lockedUntil - now) / 1000) };
  }
  if (lockedUntil && now >= lockedUntil) {
    cache.remove(keys.locked);
    cache.remove(keys.attempts);
  }
  return { locked: false, remaining_seconds: 0 };
}

function registerAdminFailedAttempt(clientId) {
  var cache = CacheService.getScriptCache();
  var keys = getAdminLoginCacheKeys(clientId);
  var attempts = parseInt(cache.get(keys.attempts) || '0', 10) + 1;

  if (attempts >= ADMIN_MAX_FAILED_ATTEMPTS) {
    var lockedUntil = new Date().getTime() + (ADMIN_LOCK_SECONDS * 1000);
    cache.put(keys.locked, String(lockedUntil), ADMIN_LOCK_SECONDS);
    cache.put(keys.attempts, String(attempts), ADMIN_LOCK_SECONDS);
    return {
      status: 'locked',
      code: 'ADMIN_LOCKED',
      remaining_seconds: ADMIN_LOCK_SECONDS,
      remaining_attempts: 0,
      message: 'Login Admin terkunci sementara karena PIN salah 3 kali. Coba lagi dalam 10 menit.'
    };
  }

  cache.put(keys.attempts, String(attempts), ADMIN_LOCK_SECONDS);
  return {
    status: 'error',
    code: 'ADMIN_PIN_WRONG',
    remaining_attempts: ADMIN_MAX_FAILED_ATTEMPTS - attempts,
    message: 'PIN Admin salah.'
  };
}

function clearAdminFailedAttempts(clientId) {
  var cache = CacheService.getScriptCache();
  var keys = getAdminLoginCacheKeys(clientId);
  cache.remove(keys.attempts);
  cache.remove(keys.locked);
}

function getSpreadsheetSafe() {
  // Web App tidak selalu memiliki "active spreadsheet". Simpan ID sekali saat setup
  // agar upload tetap dapat menulis ke Sheets dan Google Drive setelah dideploy.
  var properties = PropertiesService.getScriptProperties();
  var savedId = String(properties.getProperty(NEXUS_SPREADSHEET_ID_PROPERTY) || '').trim();
  if (savedId) {
    try {
      return SpreadsheetApp.openById(savedId);
    } catch (savedSheetError) {
      properties.deleteProperty(NEXUS_SPREADSHEET_ID_PROPERTY);
    }
  }

  var active = SpreadsheetApp.getActiveSpreadsheet();
  if (active) {
    properties.setProperty(NEXUS_SPREADSHEET_ID_PROPERTY, active.getId());
    return active;
  }

  throw new Error('DATABASE_TIDAK_TERHUBUNG: Jalankan setupNexusGallery() sekali dari editor Apps Script yang terikat dengan Google Sheet database Anda, lalu deploy ulang Web App.');
}

function configureNexusSpreadsheet(spreadsheetId) {
  var id = String(spreadsheetId || '').trim();
  if (!id) throw new Error('Spreadsheet ID wajib diisi.');
  var spreadsheet = SpreadsheetApp.openById(id);
  PropertiesService.getScriptProperties().setProperty(NEXUS_SPREADSHEET_ID_PROPERTY, spreadsheet.getId());
  return { status: 'success', spreadsheet_id: spreadsheet.getId(), spreadsheet_name: spreadsheet.getName() };
}

function normalizeHeader(value) {

function normalizePinValue(value) {
  var text = String(value === null || value === undefined ? '' : value).trim();
  // Hapus apostrof Google Sheet, spasi tersembunyi, pemisah ribuan, dan desimal .0/,0.
  text = text.replace(/^'/, '').replace(/[\u200B-\u200D\uFEFF]/g, '').replace(/\s+/g, '');
  text = text.replace(/[.,]0+$/, '');
  // PIN Nexus memakai angka. Bila Google Sheet menampilkan 123.456 atau 123,456, jadikan 123456.
  return text.replace(/[^0-9]/g, '');
}

function getAdminPinValue() {
  var saved = getSettingValue('pin_admin', '123456');
  var normalized = normalizePinValue(saved);
  return normalized || '123456';
}


function normalizePhoneForLogin(value) {
  var digits = String(value || '').replace(/[^0-9]/g, '');
  if (digits.charAt(0) === '0') digits = '62' + digits.substring(1);
  return digits;
}

function getDefaultPasisPin(nohp) {

function ensureHeaders(sheet, requiredHeaders) {
  if (!sheet) throw new Error("Sheet tidak valid.");

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(requiredHeaders);
    return sheet;
  }

  var lastColumn = Math.max(sheet.getLastColumn(), 1);
  var headerValues = sheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  var existing = {};
  headerValues.forEach(function(header) {
    existing[normalizeHeader(header)] = true;
  });

  requiredHeaders.forEach(function(header) {
    if (!existing[normalizeHeader(header)]) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(header);
      existing[normalizeHeader(header)] = true;
    }
  });

  return sheet;
}

}

function parseTagihanNominal(value) {
  var digits = String(value === undefined || value === null ? '' : value).replace(/[^0-9]/g, '');
  return digits ? parseInt(digits, 10) : 0;
}

function normalizeTagihanJenis(value) {
  var normalized = normalizeHeader(value).replace(/^uang\s+/, '');
  if (normalized === 'wisuda') return 'Wisuda';
  if (normalized === 'absensi') return 'Absensi';
  throw new Error('Jenis tagihan harus Wisuda atau Absensi.');
}

function getTagihanTambahanSisa(row) {
  return Math.max(parseTagihanNominal(row && row.nominal_tagihan) - parseTagihanNominal(row && row.nominal_dibayar), 0);
}

function getTagihanTambahanStatus(row) {
  if (normalizeHeader(row && row.status) === 'dibatalkan') return 'Dibatalkan';
  var total = parseTagihanNominal(row && row.nominal_tagihan);
  var paid = parseTagihanNominal(row && row.nominal_dibayar);
  if (total > 0 && paid >= total) return 'Lunas';
  if (paid > 0) return 'Cicilan';
  return 'Belum Bayar';
}

function sanitizeTagihanTambahanForClient(row) {

    nominal_dibayar: parseTagihanNominal(row.nominal_dibayar),

function getTagihanTambahanForSession(session) {
  var rows = sheetToJSON(getTagihanTambahanSheet()).map(function(row) {
    return sanitizeTagihanTambahanForClient(row);
  });

  if (session.role !== 'admin') {
    var targetPhone = normalizePhoneForLogin(session.nohp);
    rows = rows.filter(function(row) {
      return normalizePhoneForLogin(row.nohp) === targetPhone;
    });
  }

  return rows.sort(function(a, b) {
    return String(b.dibuat_at || b.tanggal_tagihan || '').localeCompare(String(a.dibuat_at || a.tanggal_tagihan || ''));
  });
}

function getPasisTargetsForTagihan(data) {
  var input = data || {};
  var scope = normalizeHeader(input.cakupan || input.scope || 'semua');
  var allPasis = getPublicPasisData().filter(function(row) { return row && row.nohp; });

  if (scope === 'semua' || scope === 'all' || scope === 'all_pasis') {
    return allPasis;
  }

  if (scope === 'tertentu' || scope === 'selected' || scope === 'pasis_tertentu') {
    var selected = input.nohp_list || input.pasis_nohp || [];
    if (!Array.isArray(selected)) selected = [selected];
    var selectedMap = {};
    selected.forEach(function(nohp) {
      var key = normalizePhoneForLogin(nohp);
      if (key) selectedMap[key] = true;
    });

    var targets = allPasis.filter(function(row) {
      return !!selectedMap[normalizePhoneForLogin(row.nohp)];
    });
    if (targets.length === 0) throw new Error('Pilih minimal satu pasis untuk tagihan khusus.');
    return targets;
  }

  throw new Error('Cakupan tagihan tidak valid.');
}

function createTagihanTambahan(session, data) {
  var input = data || {};
  var jenis = normalizeTagihanJenis(input.jenis);
  var namaTagihan = String(input.nama_tagihan || ('Uang ' + jenis)).replace(/[\r\n]+/g, ' ').trim().substring(0, 100);
  if (!namaTagihan) namaTagihan = 'Uang ' + jenis;

  var nominal = parseTagihanNominal(input.nominal_tagihan);
  if (nominal <= 0) throw new Error('Nominal tagihan harus lebih dari Rp 0.');

  var tanggalTagihan = String(input.tanggal_tagihan || Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Jakarta', 'yyyy-MM-dd')).substring(0, 10);
  var jatuhTempo = String(input.jatuh_tempo || '').substring(0, 10);
  var keterangan = String(input.keterangan || '').replace(/[\r\n]+/g, ' ').trim().substring(0, 500);
  var targets = getPasisTargetsForTagihan(input);
  var sheet = getTagihanTambahanSheet();
  var existing = sheetToJSON(sheet);
  var batchId = Utilities.getUuid();
  var created = [];
  var skipped = [];

  targets.forEach(function(pasis) {
    var nohp = normalizePhoneForLogin(pasis.nohp);
    var duplicate = existing.some(function(row) {
      return normalizePhoneForLogin(row.nohp) === nohp &&
        normalizeHeader(row.jenis) === normalizeHeader(jenis) &&
        normalizeHeader(row.nama_tagihan) === normalizeHeader(namaTagihan) &&
        String(row.tanggal_tagihan || '').substring(0, 10) === tanggalTagihan &&
        normalizeHeader(row.status) !== 'dibatalkan';
    });

    if (duplicate) {
      skipped.push(pasis.nama || pasis.nohp);
      return;
    }

    var item = {
      id: Utilities.getUuid(),
      batch_id: batchId,
      jenis: jenis,
      nama_tagihan: namaTagihan,
      nohp: pasis.nohp || '',
      nama: pasis.nama || '',
      nominal_tagihan: nominal,
      nominal_dibayar: 0,
      tanggal_tagihan: tanggalTagihan,
      jatuh_tempo: jatuhTempo,
      keterangan: keterangan,
      riwayat: '',
      status: 'Belum Bayar',
      dibuat_oleh: session.nama || 'Admin',
      dibuat_at: nowString(),
      updated_at: nowString()
    };
    appendByHeaders(sheet, item);
    existing.push(item);
    created.push(sanitizeTagihanTambahanForClient(item));
  });

  if (created.length === 0) {
    throw new Error('Tidak ada tagihan dibuat. Seluruh pasis pilihan sudah memiliki tagihan yang sama.');
  }

  return {
    status: 'success',
    data: created,
    batch_id: batchId,
    created_count: created.length,
    skipped_count: skipped.length,
    skipped_names: skipped.slice(0, 10),
    message: created.length + ' tagihan ' + jenis + ' berhasil dibuat' + (skipped.length ? '. ' + skipped.length + ' pasis dilewati karena tagihan duplikat.' : '.')
  };
}

function bayarTagihanTambahan(data) {
  var input = data || {};
  var itemId = String(input.id || '').trim();
  if (!itemId) throw new Error('ID tagihan tidak tersedia.');

  var nominalBayar = parseTagihanNominal(input.nominal);
  if (nominalBayar <= 0) throw new Error('Nominal pembayaran harus lebih dari Rp 0.');

  var sheet = getTagihanTambahanSheet();
  var rowIndex = findRowByHeaderValue(sheet, 'id', itemId);
  if (rowIndex === -1) throw new Error('Tagihan tidak ditemukan.');

  var headers = getHeaderInfo(sheet).headers;
  var values = sheet.getRange(rowIndex, 1, 1, headers.length).getDisplayValues()[0];
  var row = rowToObjectFromValues(headers, values);
  if (normalizeHeader(row.status) === 'dibatalkan') throw new Error('Tagihan yang dibatalkan tidak dapat menerima pembayaran.');

  var sisa = getTagihanTambahanSisa(row);
  if (sisa <= 0) throw new Error('Tagihan ini sudah lunas.');
  if (nominalBayar > sisa) throw new Error('Nominal melebihi sisa tagihan (' + sisa + ').');

  var tanggal = String(input.tanggal || Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Jakarta', 'yyyy-MM-dd')).substring(0, 10);
  var catatan = String(input.catatan || '').replace(/[\r\n]+/g, ' ').trim().substring(0, 250);
  var dibayarBaru = parseTagihanNominal(row.nominal_dibayar) + nominalBayar;
  var riwayatBaru = String(row.riwayat || '');
  var riwayatItem = tanggal + '::' + nominalBayar + (catatan ? '::' + catatan : '');
  riwayatBaru = riwayatBaru ? riwayatBaru + '||' + riwayatItem : riwayatItem;
  var statusBaru = dibayarBaru >= parseTagihanNominal(row.nominal_tagihan) ? 'Lunas' : 'Cicilan';

  writeByHeaders(sheet, rowIndex, {
    nominal_dibayar: dibayarBaru,
    riwayat: riwayatBaru,
    status: statusBaru,
    updated_at: nowString()
  });

  row.nominal_dibayar = dibayarBaru;
  row.riwayat = riwayatBaru;
  row.status = statusBaru;
  row.updated_at = nowString();

  return {
    status: 'success',
    data: sanitizeTagihanTambahanForClient(row),
    message: statusBaru === 'Lunas' ? 'Pembayaran dicatat. Tagihan sudah lunas.' : 'Pembayaran dicatat. Sisa tagihan masih tersedia.'
  };
}

function gallerySafeText(value, fallback, maxLength) {
  var text = String(value === undefined || value === null ? '' : value)
    .replace(/[\r\n]+/g, ' ')
    .replace(/[^a-zA-Z0-9À-ÿ _.,()&+\-]/g, '')
    .trim();
  if (!text) text = fallback || '';
  return text.substring(0, maxLength || 120);
}

function parseGalleryImageData(dataUrl) {
  var raw = String(dataUrl || '');
  var match = raw.match(/^data:(image\/(?:jpeg|jpg|png|webp));base64,([A-Za-z0-9+/=\s]+)$/i);
  if (!match) throw new Error('Format foto tidak didukung. Gunakan JPG, PNG, atau WEBP.');

  var mimeType = match[1].toLowerCase().replace('image/jpg', 'image/jpeg');
  var bytes = Utilities.base64Decode(match[2].replace(/\s/g, ''));
  if (!bytes || bytes.length === 0) throw new Error('Data foto kosong.');
  if (bytes.length > GALLERY_MAX_FILE_BYTES) throw new Error('Ukuran satu foto maksimal 4 MB setelah dikompresi.');

  return { mime_type: mimeType, bytes: bytes };
}

function getGalleryExtension(mimeType) {
  if (mimeType === 'image/png') return 'png';
  if (mimeType === 'image/webp') return 'webp';
  return 'jpg';
}

function makeGalleryFileName(title, index, mimeType) {
  var stamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Jakarta', 'yyyyMMdd_HHmmss');
  var safeTitle = gallerySafeText(title, 'foto', 50).replace(/\s+/g, '-').toLowerCase();
  return stamp + '_' + safeTitle + '_' + (index + 1) + '.' + getGalleryExtension(mimeType);
}

function createGalleryFiles(session, data) {
  var input = data || {};
  var images = input.images || [];
  if (!Array.isArray(images) || images.length === 0) throw new Error('Pilih minimal satu foto untuk diunggah.');
  if (images.length > GALLERY_MAX_UPLOAD_COUNT) throw new Error('Maksimal ' + GALLERY_MAX_UPLOAD_COUNT + ' foto per unggahan.');

  var title = gallerySafeText(input.judul, 'Dokumentasi Kegiatan', 100);
  var caption = String(input.caption || '').replace(/[\r\n]+/g, ' ').trim().substring(0, 500);
  var album = gallerySafeText(input.album, 'Dokumentasi', 70);
  var category = gallerySafeText(input.kategori, 'Kegiatan', 40);
  var dateValue = String(input.tanggal || Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Jakarta', 'yyyy-MM-dd')).substring(0, 10);
  var folder = getOrCreateGalleryAlbumFolder(album, dateValue);
  var gallerySheet = getGaleriSheet();
  var created = [];
  var sharingWarning = false;

  images.forEach(function(image, index) {
    var file = null;
    try {
      var parsed = parseGalleryImageData(image && image.data_url);
      var fileName = makeGalleryFileName(title, index, parsed.mime_type);
      var blob = Utilities.newBlob(parsed.bytes, parsed.mime_type, fileName);
      file = folder.createFile(blob);
      file.setDescription('Nexus Gallery | ' + title + ' | ' + album + ' | ' + nowString());

      try {
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      } catch (sharingError) {
        // Beberapa Google Workspace menonaktifkan sharing publik. File tetap tersimpan;
        // admin dapat mengatur sharing Drive secara manual bila preview tidak muncul.
        sharingWarning = true;
      }

      var links = makeGalleryPublicLinks(file.getId());
      var row = {
        id: Utilities.getUuid(),
        judul: images.length > 1 ? title + ' (' + (index + 1) + ')' : title,
        caption: caption,
        album: album,
        kategori: category,
        tanggal: dateValue,
        file_id: file.getId(),
        file_name: file.getName(),
        mime_type: parsed.mime_type,
        size_bytes: file.getSize(),
        url_view: links.url_view,
        url_download: links.url_download,
        thumbnail_url: links.thumbnail_url,
        folder_id: folder.getId(),
        uploaded_by: session.nama || 'Admin',
        uploaded_at: nowString(),
        status: 'Aktif',
        deleted_at: ''
      };

      appendByHeaders(gallerySheet, row);
      created.push(row);
    } catch (uploadError) {
      // Hindari file Drive tanpa baris metadata apabila transaksi gagal di tengah proses.
      if (file) {
        try { file.setTrashed(true); } catch (cleanupError) {}
      }
      throw uploadError;
    }
  });

  return {
    status: 'success',
    data: created,
    folder_url: folder.getUrl(),
    sharing_warning: sharingWarning,
    message: created.length + ' foto berhasil tersimpan di Google Drive.'
  };
}

function buildPhoneCandidates(value) {
  var raw = String(value || '').replace(/^'/, '').trim();
  var digits = raw.replace(/[^0-9]/g, '');
  var normalized = normalizePhoneForLogin(raw);
  var candidates = {};
  [raw, digits, normalized].forEach(function(item) {
    if (item) candidates[item] = true;
  });
  if (normalized.indexOf('62') === 0) candidates['0' + normalized.substring(2)] = true;
  if (digits.charAt(0) === '0') candidates['62' + digits.substring(1)] = true;
  return Object.keys(candidates);
}

function findRowByHeaderValue(sheet, headerName, targetValue) {
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return -1;

  var headers = data[0].map(normalizeHeader);
  var targetIndex = headers.indexOf(normalizeHeader(headerName));
  if (targetIndex === -1) throw new Error("Kolom '" + headerName + "' tidak ditemukan.");

  var target = String(targetValue || '').replace(/^'/, '').trim();
  for (var r = 1; r < data.length; r++) {
    var cell = String(data[r][targetIndex] || '').replace(/^'/, '').trim();
    if (cell === target) return r + 1;
  }

  return -1;
}



function findPasisRowByPhone(sheet, nohp) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return { rowIndex: -1, row: null };

  var info = getHeaderInfo(sheet);
  var headers = info.headers;
  var nohpIndex = info.map.nohp;
  if (nohpIndex === undefined) throw new Error("Kolom 'nohp' tidak ditemukan.");

  var target = normalizePhoneForLogin(nohp);
  var phoneRange = sheet.getRange(2, nohpIndex + 1, lastRow - 1, 1);
  var candidates = buildPhoneCandidates(nohp);

  // Jalur cepat: cari exact match di kolom No HP, lalu validasi normalisasi.
  for (var i = 0; i < candidates.length; i++) {
    var finder = phoneRange.createTextFinder(candidates[i]).matchEntireCell(true);
    var foundCell = finder.findNext();
    if (foundCell) {
      var rowIndex = foundCell.getRow();
      var rowValues = sheet.getRange(rowIndex, 1, 1, info.lastColumn).getDisplayValues()[0];
      if (normalizePhoneForLogin(rowValues[nohpIndex]) === target) {
        return { rowIndex: rowIndex, row: rowToObjectFromValues(headers, rowValues) };
      }
    }
  }

  // Fallback tetap aman untuk format nomor yang berisi spasi/tanda hubung.
  var phoneValues = phoneRange.getDisplayValues();
  for (var r = 0; r < phoneValues.length; r++) {
    if (normalizePhoneForLogin(phoneValues[r][0]) === target) {
      var fallbackRowIndex = r + 2;
      var fallbackValues = sheet.getRange(fallbackRowIndex, 1, 1, info.lastColumn).getDisplayValues()[0];
      return { rowIndex: fallbackRowIndex, row: rowToObjectFromValues(headers, fallbackValues) };
    }
  }

  return { rowIndex: -1, row: null };
}

function writeByHeaders(sheet, rowIndex, payloadData) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];

  headers.forEach(function(header, idx) {
    var key = normalizeHeader(header);
    if (payloadData[key] !== undefined) {
      var val = payloadData[key];
      if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
      sheet.getRange(rowIndex, idx + 1).setValue(val);
    }
  });
}

function appendByHeaders(sheet, payloadData) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
  var newRow = [];

  headers.forEach(function(header) {
    var key = normalizeHeader(header);
    var val = payloadData[key] !== undefined ? payloadData[key] : "";
    if (key === 'id' && val === "") val = new Date().getTime().toString();
    if (key === 'pin_diubah' && val === "") val = "Belum";
    if (key === 'pin_wa_terkirim' && val === "") val = "Belum";
    if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
    newRow.push(val);
  });

  sheet.appendRow(newRow);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents || '{}');
    if (!payload.action) throw new Error("Action kosong.");

    // --- AMBIL DATA SETELAH LOGIN ---
    if (payload.action === 'get_secure_data') {
      var authSession = requireAuthSession(payload);
      var secureResult = getSecureDataForSession(authSession);
      return makeOutput(secureResult);
    }

    // --- TAGIHAN TAMBAHAN: UANG WISUDA & UANG ABSENSI ---
    if (payload.action === 'create_tagihan_tambahan') {
      var tagihanAdmin = requireAdminSession(payload);
      return makeOutput(createTagihanTambahan(tagihanAdmin, payload.data || {}));
    }

    if (payload.action === 'bayar_tagihan_tambahan') {
      requireAdminSession(payload);
      return makeOutput(bayarTagihanTambahan(payload.data || {}));
    }

    if (payload.action === 'batalkan_tagihan_tambahan') {
      requireAdminSession(payload);
      return makeOutput(batalkanTagihanTambahan(payload.data || {}));
    }

    // --- GALERI GOOGLE DRIVE ---
    if (payload.action === 'initialize_gallery') {
      requireAdminSession(payload);
      return makeOutput(setupNexusGallery());
    }

    if (payload.action === 'list_gallery') {
      var gallerySession = requireAuthSession(payload);
      ensureGalleryAccess(gallerySession);
      return makeOutput(getGalleryListForSession(gallerySession, payload.data || {}));
    }

    if (payload.action === 'upload_gallery') {
      var galleryAdminSession = requireAdminSession(payload);
      return makeOutput(createGalleryFiles(galleryAdminSession, payload.data || {}));
    }

    if (payload.action === 'delete_gallery') {
      requireAdminSession(payload);
      return makeOutput(deleteGalleryItem(payload.data || {}));
    }

    // --- LOGIN ADMIN CEPAT + PROTEKSI 3x SALAH PER PERANGKAT/BROWSER ---
    if (payload.action === 'login_admin') {
      var adminData = payload.data || {};
      var pinAdmin = normalizePinValue(adminData.pin);
      var clientId = adminData.client_id || adminData.device_id || 'global';
      if (!pinAdmin) throw new Error("PIN Admin wajib diisi.");

      var savedAdminPin = getAdminPinValue();

      // PIN benar harus tetap bisa masuk walaupun perangkat sempat terkunci karena salah input.
      // Ini mencegah kasus admin asli tidak bisa login karena lock lama/cache browser.
      if (pinAdmin === savedAdminPin) {
        clearAdminFailedAttempts(clientId);
        var settingsLogin = getSettingsObject();
        return makeOutput({
          status: 'success',
          role: 'admin',
          token: createAuthSession('admin', '', 'Admin'),
          pengaturan: Object.keys(settingsLogin).map(function(key) {
            return { key: key, value: settingsLogin[key] };
          }),
          message: 'Login Admin berhasil.'
        });
      }

      var lockState = getAdminLockState(clientId);
      if (lockState.locked) {
        return makeOutput({
          status: 'locked',
          code: 'ADMIN_LOCKED',
          remaining_seconds: lockState.remaining_seconds,
          remaining_attempts: 0,
          message: 'Login Admin terkunci sementara karena PIN salah 3 kali. Coba lagi dalam ' + Math.ceil(lockState.remaining_seconds / 60) + ' menit.'
        });
      }

      return makeOutput(registerAdminFailedAttempt(clientId));
    }

    // --- LOGIN PASIS ---
    if (payload.action === 'login_pasis') {
      var loginData = payload.data || {};
      var nohpLogin = loginData.nohp;
      var pinLogin = String(loginData.pin || '').trim();
      if (!nohpLogin || !pinLogin) throw new Error("No HP dan PIN wajib diisi.");

      var sheetLogin = getDataPasisSheet();
      var foundLogin = findPasisRowByPhone(sheetLogin, nohpLogin);
      if (foundLogin.rowIndex === -1 || !foundLogin.row) throw new Error("Akun pasis tidak ditemukan.");

      var effectivePin = getEffectivePasisPin(foundLogin.row);
      if (pinLogin !== effectivePin) throw new Error("PIN salah.");

      var loginTime = nowString();
      writeByHeaders(sheetLogin, foundLogin.rowIndex, { last_login: loginTime });
      foundLogin.row.last_login = loginTime;

      var loginIsCuti = isPasisCuti(foundLogin.row);
      return makeOutput({
        status: 'success',
        role: 'pasis',
        token: createAuthSession('pasis', normalizePhoneForLogin(foundLogin.row.nohp), foundLogin.row.nama || 'Pasis'),
        user: sanitizePasisForClient(foundLogin.row),
        pasis_status: loginIsCuti ? 'cuti' : 'aktif',
        access_scope: loginIsCuti ? 'self' : 'all',
        access_views: loginIsCuti ? ['pasis', 'akun'] : ['dashboard', 'pasis', 'akun', 'pengeluaran', 'gallery'],
        default_pin_hint: !foundLogin.row.pin_login ? 'PIN awal adalah 6 digit terakhir No HP.' : '',
        message: loginIsCuti ? 'Login berhasil. Status Cuti: akses hanya Data Pasis milik Anda dan Akun Saya.' : 'Login berhasil.'
      });
    }

    // --- GANTI PIN PASIS ---
    if (payload.action === 'update_pin_pasis') {
      var pinData = payload.data || {};
      var nohpPin = pinData.nohp;
      var oldPin = String(pinData.old_pin || '').trim();
      var newPin = String(pinData.new_pin || '').trim();

      if (!nohpPin || !oldPin || !newPin) throw new Error("No HP, PIN lama, dan PIN baru wajib diisi.");
      if (!/^\d{4,12}$/.test(newPin)) throw new Error("PIN baru harus angka 4 sampai 12 digit.");

      var sheetPin = getDataPasisSheet();
      var foundPin = findPasisRowByPhone(sheetPin, nohpPin);
      if (foundPin.rowIndex === -1 || !foundPin.row) throw new Error("Akun pasis tidak ditemukan.");

      var currentPin = getEffectivePasisPin(foundPin.row);
      if (oldPin !== currentPin) throw new Error("PIN lama salah.");
      if (newPin === currentPin) throw new Error("PIN baru tidak boleh sama dengan PIN lama.");

      writeByHeaders(sheetPin, foundPin.rowIndex, {
        pin_login: newPin,
        pin_diubah: 'Ya',
        last_pin_change: nowString()
      });

      foundPin.row.pin_diubah = 'Ya';
      foundPin.row.last_pin_change = nowString();
      return makeOutput({
        status: 'success',
        message: 'PIN berhasil diperbarui.',
        user: sanitizePasisForClient(foundPin.row)
      });
    }

    // --- SHARE PIN PASIS VIA WHATSAPP (ADMIN) ---
    if (payload.action === 'get_pin_share_list') {
      requireAdminSession(payload);
      return makeOutput({
        status: 'success',
        data: getPinShareDataForAdmin()
      });
    }

    if (payload.action === 'generate_share_pin' || payload.action === 'reset_active_pasis_pins') {
      requireAdminSession(payload);
      var resetList = generatePinsForSharePinPasis();
      return makeOutput({
        status: 'success',
        updated: resetList.length,
        data: resetList,
        message: resetList.length + ' PIN pasis Aktif/Cuti berhasil dibuat ulang.'
      });
    }

    if (payload.action === 'mark_pin_shared') {
      requireAdminSession(payload);
      var shareData = payload.data || {};
      var listPhone = shareData.nohp_list || [];
      if (!Array.isArray(listPhone)) listPhone = [];
      if (shareData.nohp) listPhone.push(shareData.nohp);
      if (listPhone.length === 0) throw new Error("Daftar No HP kosong.");

      var sheetShare = getDataPasisSheet();
      var updatedCount = 0;
      listPhone.forEach(function(phone) {
        if (markPinSharedByPhone(sheetShare, phone)) updatedCount++;
      });

      return makeOutput({
        status: 'success',
        updated: updatedCount,
        message: updatedCount + ' status share PIN diperbarui.'
      });
    }

    // --- AKSI PING (KUNJUNGAN & USER AKTIF) ---
    if (payload.action === 'ping') {
      var cache = CacheService.getScriptCache();
      var activeUsersStr = cache.get('nexus_active_users');
      var activeUsers = activeUsersStr ? JSON.parse(activeUsersStr) : {};
      var now = new Date().getTime();

      if (payload.sessionId) activeUsers[payload.sessionId] = now;

      var activeCount = 0;
      for (var key in activeUsers) {
        if (now - activeUsers[key] > 120000) delete activeUsers[key];
        else activeCount++;
      }
      cache.put('nexus_active_users', JSON.stringify(activeUsers), 300);

      var sheetPengaturan = getPengaturanSheet();
      var dataPengaturan = sheetPengaturan.getDataRange().getDisplayValues();
      var totalViews = 0;
      var viewRowIndex = -1;

      for (var r = 1; r < dataPengaturan.length; r++) {
        if (normalizeHeader(dataPengaturan[r][0]) === 'total_views') {
          viewRowIndex = r + 1;
          totalViews = parseInt(dataPengaturan[r][1]) || 0;
          break;
        }
      }

      if (payload.isNewVisit) {
        totalViews++;
        if (viewRowIndex > -1) sheetPengaturan.getRange(viewRowIndex, 2).setValue(totalViews);
        else sheetPengaturan.appendRow(['total_views', totalViews]);
      }

      return makeOutput({ status: 'success', activeUsers: activeCount, totalViews: totalViews });
    }

    // --- AKSI DATA PASIS ---
    if (payload.action === 'create' || payload.action === 'update') {
      requireAdminSession(payload);
      var sheetPasis = getDataPasisSheet();

      if (payload.action === 'create') {
        var createData = payload.data || {};
        if (!createData.pin_diubah) createData.pin_diubah = 'Belum';
        appendByHeaders(sheetPasis, createData);
        sortDataRowsByFirstColumn(sheetPasis);
        return makeOutput({ status: 'success' });
      }

      if (payload.action === 'update') {
        var data = payload.data || {};
        var targetNohp = data.original_nohp || data.nohp;
        if (!targetNohp) throw new Error("Nomor HP target kosong.");

        var rowIndex = findRowByHeaderValue(sheetPasis, 'nohp', targetNohp);
        if (rowIndex === -1) throw new Error("Data tidak ditemukan.");

        writeByHeaders(sheetPasis, rowIndex, data);
        return makeOutput({ status: 'success' });
      }
    }

    if (payload.action === 'delete_pasis') {
      requireAdminSession(payload);
      var sheetDeletePasis = getDataPasisSheet();
      var rowDelete = findRowByHeaderValue(sheetDeletePasis, 'nohp', payload.data && payload.data.nohp);
      if (rowDelete === -1) throw new Error("Pasis tidak ditemukan.");

      sheetDeletePasis.deleteRow(rowDelete);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI PENGELUARAN ---
    if (payload.action === 'create_pengeluaran') {
      requireAdminSession(payload);
      appendByHeaders(getPengeluaranSheet(), payload.data || {});
      return makeOutput({ status: 'success' });
    }

    if (payload.action === 'delete_pengeluaran') {
      requireAdminSession(payload);
      var sheetPengeluaran = getPengeluaranSheet();
      var rowPengeluaran = findRowByHeaderValue(sheetPengeluaran, 'id', payload.data && payload.data.id);
      if (rowPengeluaran === -1) throw new Error("Pengeluaran tidak ditemukan.");

      sheetPengeluaran.deleteRow(rowPengeluaran);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI PINDAH DANA / SERAH TERIMA KE KETUA KELAS ---
    if (payload.action === 'create_pindah_dana') {
      requireAdminSession(payload);
      appendByHeaders(getPindahDanaSheet(), payload.data || {});
      return makeOutput({ status: 'success' });
    }

    if (payload.action === 'delete_pindah_dana') {
      requireAdminSession(payload);
      var sheetPindah = getPindahDanaSheet();
      var rowPindah = findRowByHeaderValue(sheetPindah, 'id', payload.data && payload.data.id);
      if (rowPindah === -1) throw new Error("Data serah terima kas tidak ditemukan.");

      sheetPindah.deleteRow(rowPindah);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI UPDATE PENGATURAN ---
    if (payload.action === 'update_pengaturan') {
      requireAdminSession(payload);
      var sheetSetting = getPengaturanSheet();
      var settingData = sheetSetting.getDataRange().getDisplayValues();
      var keysToUpdate = payload.data || {};

      for (var k in keysToUpdate) {
        var found = false;
        for (var s = 1; s < settingData.length; s++) {
          if (normalizeHeader(settingData[s][0]) === normalizeHeader(k)) {
            sheetSetting.getRange(s + 1, 2).setValue(keysToUpdate[k]);
            found = true;
            break;
          }
        }
        if (!found) sheetSetting.appendRow([k, keysToUpdate[k]]);
      }

      return makeOutput({ status: 'success' });
    }

    throw new Error("Action tidak valid: " + payload.action);
  } catch (err) {
    var message = err && err.message ? err.message : String(err || 'Kesalahan tidak diketahui.');
    console.error('Nexus API error: ' + message + '\n' + (err && err.stack ? err.stack : ''));
    return makeOutput({ status: 'error', message: message });
  }
}
