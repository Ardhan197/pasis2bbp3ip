var DEFAULT_SETTINGS = {
  target_tagihan: 15000000,
  target_cuti_refund: 0,
  pin_admin: '123456'
};

var AUTH_SESSION_TTL_SECONDS = 6 * 60 * 60;
var ADMIN_MAX_FAILED_ATTEMPTS = 3;
var ADMIN_LOCK_SECONDS = 10 * 60;

function normalizeClientId(value) {
  var id = String(value || 'global').replace(/[^a-zA-Z0-9_\-]/g, '').substring(0, 80);
  return id || 'global';
}

function getAdminLoginCacheKeys(clientId) {
  var id = normalizeClientId(clientId);
  return { attempts: 'nexus_admin_attempts_' + id, locked: 'nexus_admin_locked_' + id };
}

function getAdminLockState(clientId) {
  var cache = CacheService.getScriptCache();
  var keys = getAdminLoginCacheKeys(clientId);
  var lockedUntil = parseInt(cache.get(keys.locked) || '0', 10);
  var now = new Date().getTime();
  if (lockedUntil && now < lockedUntil) {
    return { locked: true, remaining_seconds: Math.ceil((lockedUntil - now) / 1000) };
  }
  if (lockedUntil && now >= lockedUntil) {
    cache.remove(keys.locked);
    cache.remove(keys.attempts);
  }
  return { locked: false, remaining_seconds: 0 };
}

function registerAdminFailedAttempt(clientId) {
  var cache = CacheService.getScriptCache();
  var keys = getAdminLoginCacheKeys(clientId);
  var attempts = parseInt(cache.get(keys.attempts) || '0', 10) + 1;

  if (attempts >= ADMIN_MAX_FAILED_ATTEMPTS) {
    var lockedUntil = new Date().getTime() + (ADMIN_LOCK_SECONDS * 1000);
    cache.put(keys.locked, String(lockedUntil), ADMIN_LOCK_SECONDS);
    cache.put(keys.attempts, String(attempts), ADMIN_LOCK_SECONDS);
    return {
      status: 'locked',
      code: 'ADMIN_LOCKED',
      remaining_seconds: ADMIN_LOCK_SECONDS,
      remaining_attempts: 0,
      message: 'Login Admin terkunci sementara karena PIN salah 3 kali. Coba lagi dalam 10 menit.'
    };
  }

  cache.put(keys.attempts, String(attempts), ADMIN_LOCK_SECONDS);
  return {
    status: 'error',
    code: 'ADMIN_PIN_WRONG',
    remaining_attempts: ADMIN_MAX_FAILED_ATTEMPTS - attempts,
    message: 'PIN Admin salah.'
  };
}

function clearAdminFailedAttempts(clientId) {
  var cache = CacheService.getScriptCache();
  var keys = getAdminLoginCacheKeys(clientId);
  cache.remove(keys.attempts);
  cache.remove(keys.locked);
}

function getSpreadsheetSafe() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error("Koneksi ditolak. Pastikan script ditempel pada Apps Script milik Google Sheets.");
  return ss;
}

function normalizeHeader(value) {
  return String(value || '').toLowerCase().trim();
}


function normalizePinValue(value) {
  var text = String(value === null || value === undefined ? '' : value).trim();
  // Hapus apostrof Google Sheet, spasi tersembunyi, pemisah ribuan, dan desimal .0/,0.
  text = text.replace(/^'/, '').replace(/[\u200B-\u200D\uFEFF]/g, '').replace(/\s+/g, '');
  text = text.replace(/[.,]0+$/, '');
  // PIN Nexus memakai angka. Bila Google Sheet menampilkan 123.456 atau 123,456, jadikan 123456.
  return text.replace(/[^0-9]/g, '');
}

function getAdminPinValue() {
  var saved = getSettingValue('pin_admin', '123456');
  var normalized = normalizePinValue(saved);
  return normalized || '123456';
}


function normalizePhoneForLogin(value) {
  var digits = String(value || '').replace(/[^0-9]/g, '');
  if (digits.charAt(0) === '0') digits = '62' + digits.substring(1);
  return digits;
}

function getDefaultPasisPin(nohp) {
  var digits = normalizePhoneForLogin(nohp);
  return digits.length >= 6 ? digits.slice(-6) : digits;
}

function isPasisCuti(row) {
  return normalizeHeader(row && row.status_aktif) === 'cuti';
}

function getEffectivePasisPin(row) {
  var customPin = String((row && row.pin_login) || '').trim();
  return customPin || getDefaultPasisPin(row && row.nohp);
}

function sanitizePasisForClient(row) {
  var clean = {};
  for (var key in row) clean[key] = row[key];
  delete clean.pin_login;
  return clean;
}

function nowString() {
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss');
}

function ensureHeaders(sheet, requiredHeaders) {
  if (!sheet) throw new Error("Sheet tidak valid.");

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(requiredHeaders);
    return sheet;
  }

  var lastColumn = Math.max(sheet.getLastColumn(), 1);
  var headerValues = sheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  var existing = {};
  headerValues.forEach(function(header) {
    existing[normalizeHeader(header)] = true;
  });

  requiredHeaders.forEach(function(header) {
    if (!existing[normalizeHeader(header)]) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(header);
      existing[normalizeHeader(header)] = true;
    }
  });

  return sheet;
}

function getOrCreateSheet(name, headers) {
  var ss = getSpreadsheetSafe();
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return ensureHeaders(sheet, headers);
}

function getSheetSafe(name) {
  var ss = getSpreadsheetSafe();
  var sheet = ss.getSheetByName(name);
  if (!sheet) throw new Error("Tab '" + name + "' tidak ditemukan.");
  return sheet;
}

  return getOrCreateSheet("Data Pasis", DATA_PASIS_HEADERS);

function getPengeluaranSheet() {

function getPindahDanaSheet() {
  return getOrCreateSheet("Pindah Dana", PINDAH_DANA_HEADERS);

function getPengaturanSheet() {
  var sheet = getOrCreateSheet("Pengaturan", PENGATURAN_HEADERS);
  ensureDefaultSettings(sheet);
  return sheet;

function ensureDefaultSettings(sheet) {
  var data = sheet.getDataRange().getDisplayValues();
  var existing = {};
  for (var r = 1; r < data.length; r++) {
    existing[normalizeHeader(data[r][0])] = true;
  }

  for (var key in DEFAULT_SETTINGS) {
    if (!existing[normalizeHeader(key)]) {
      sheet.appendRow([key, DEFAULT_SETTINGS[key]]);
      existing[normalizeHeader(key)] = true;
    }
  }
}

function sheetToJSON(sheet) {
  if (!sheet) return [];
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return [];

  var headers = data[0].map(function(header) {
    return normalizeHeader(header);
  });

  return data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(header, index) {
      if (header) obj[header] = row[index];
    });
    return obj;
  }).filter(function(r) {
    var firstKey = headers[0];
    return r[firstKey] && r[firstKey] !== "";
  });
}


function getPinShareDataForAdmin() {
  return sheetToJSON(getDataPasisSheet()).filter(function(row) {
    // Pasis Aktif dan Cuti sama-sama bisa login, jadi keduanya masuk daftar share PIN.
    return row && row.nohp;
  }).map(function(row) {
    return {
      nama: row.nama || '',
      nohp: row.nohp || '',
      jk: row.jk || '',
      status_aktif: row.status_aktif || 'Aktif',
      pin: getEffectivePasisPin(row),
      pin_diubah: row.pin_diubah || 'Belum',
      pin_wa_terkirim: row.pin_wa_terkirim || 'Belum',
      last_pin_share: row.last_pin_share || '',
      last_login: row.last_login || ''
    };
  });
}


function generateRandomPasisPin() {
  var pin = '';
  for (var i = 0; i < 6; i++) {
    pin += Math.floor(Math.random() * 10).toString();
  }
  return pin;
}

function resetPinsForActivePasis() {
  var sheet = getDataPasisSheet();
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return [];

  var headers = data[0].map(normalizeHeader);
  var nohpIndex = headers.indexOf('nohp');
  if (nohpIndex === -1) throw new Error("Kolom 'nohp' tidak ditemukan.");

  var updatedRows = [];
  for (var r = 1; r < data.length; r++) {
    var rowObj = {};
    headers.forEach(function(header, index) {
      if (header) rowObj[header] = data[r][index];
    });

    // Pasis Aktif dan Cuti sama-sama bisa login, jadi PIN keduanya boleh di-reset admin.
    if (!rowObj.nohp) continue;

    var newPin = generateRandomPasisPin();
    writeByHeaders(sheet, r + 1, {
      pin_login: newPin,
      pin_diubah: 'Reset Admin',
      last_pin_change: nowString(),
      pin_wa_terkirim: 'Belum',
      last_pin_share: ''
    });

    rowObj.pin_login = newPin;
    rowObj.pin_diubah = 'Reset Admin';
    rowObj.last_pin_change = nowString();
    rowObj.pin_wa_terkirim = 'Belum';
    rowObj.last_pin_share = '';
    updatedRows.push({
      nama: rowObj.nama || '',
      nohp: rowObj.nohp || '',
      jk: rowObj.jk || '',
      status_aktif: rowObj.status_aktif || 'Aktif',
      pin: newPin,
      pin_diubah: rowObj.pin_diubah,
      pin_wa_terkirim: rowObj.pin_wa_terkirim,
      last_pin_share: rowObj.last_pin_share,
      last_login: rowObj.last_login || ''
    });
  }

  return updatedRows;
}

function generatePinsForSharePinPasis() {
  // Alias khusus fitur Share PIN agar nama action frontend lebih jelas.
  // Fungsi ini membuat PIN baru untuk semua pasis yang memiliki No HP, termasuk status Aktif dan Cuti.
  return resetPinsForActivePasis();
}

function markPinSharedByPhone(sheet, nohp) {
  var found = findPasisRowByPhone(sheet, nohp);
  if (found.rowIndex === -1 || !found.row) return false;
  writeByHeaders(sheet, found.rowIndex, {
    pin_wa_terkirim: 'Ya',
    last_pin_share: nowString()
  });
  return true;
}


function getSettingsObject() {
  var rows = sheetToJSON(getPengaturanSheet());
  var obj = {};
  rows.forEach(function(row) {
    var key = normalizeHeader(row.key);
    if (key) obj[key] = row.value;
  });
  return obj;
}

function getSettingValue(key, fallback) {
  var settings = getSettingsObject();
  var val = settings[normalizeHeader(key)];
  return val !== undefined && val !== '' ? String(val).trim() : fallback;
}

function getHeaderInfo(sheet) {
  var lastColumn = Math.max(sheet.getLastColumn(), 1);
  var headers = sheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0].map(normalizeHeader);
  var map = {};
  headers.forEach(function(header, index) {
    if (header) map[header] = index;
  });
  return { headers: headers, map: map, lastColumn: lastColumn };
}

function rowToObjectFromValues(headers, values) {
  var obj = {};
  headers.forEach(function(header, index) {
    if (header) obj[header] = values[index];
  });
  return obj;
}

function buildPhoneCandidates(value) {
  var raw = String(value || '').replace(/^'/, '').trim();
  var digits = raw.replace(/[^0-9]/g, '');
  var normalized = normalizePhoneForLogin(raw);
  var candidates = {};
  [raw, digits, normalized].forEach(function(item) {
    if (item) candidates[item] = true;
  });
  if (normalized.indexOf('62') === 0) candidates['0' + normalized.substring(2)] = true;
  if (digits.charAt(0) === '0') candidates['62' + digits.substring(1)] = true;
  return Object.keys(candidates);
}

function getPublicPasisData() {
  return sheetToJSON(getDataPasisSheet()).map(sanitizePasisForClient);
}

function makeOutput(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function findRowByHeaderValue(sheet, headerName, targetValue) {
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return -1;

  var headers = data[0].map(normalizeHeader);
  var targetIndex = headers.indexOf(normalizeHeader(headerName));
  if (targetIndex === -1) throw new Error("Kolom '" + headerName + "' tidak ditemukan.");

  var target = String(targetValue || '').replace(/^'/, '').trim();
  for (var r = 1; r < data.length; r++) {
    var cell = String(data[r][targetIndex] || '').replace(/^'/, '').trim();
    if (cell === target) return r + 1;
  }

  return -1;
}



function findPasisRowByPhone(sheet, nohp) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return { rowIndex: -1, row: null };

  var info = getHeaderInfo(sheet);
  var headers = info.headers;
  var nohpIndex = info.map.nohp;
  if (nohpIndex === undefined) throw new Error("Kolom 'nohp' tidak ditemukan.");

  var target = normalizePhoneForLogin(nohp);
  var phoneRange = sheet.getRange(2, nohpIndex + 1, lastRow - 1, 1);
  var candidates = buildPhoneCandidates(nohp);

  // Jalur cepat: cari exact match di kolom No HP, lalu validasi normalisasi.
  for (var i = 0; i < candidates.length; i++) {
    var finder = phoneRange.createTextFinder(candidates[i]).matchEntireCell(true);
    var foundCell = finder.findNext();
    if (foundCell) {
      var rowIndex = foundCell.getRow();
      var rowValues = sheet.getRange(rowIndex, 1, 1, info.lastColumn).getDisplayValues()[0];
      if (normalizePhoneForLogin(rowValues[nohpIndex]) === target) {
        return { rowIndex: rowIndex, row: rowToObjectFromValues(headers, rowValues) };
      }
    }
  }

  // Fallback tetap aman untuk format nomor yang berisi spasi/tanda hubung.
  var phoneValues = phoneRange.getDisplayValues();
  for (var r = 0; r < phoneValues.length; r++) {
    if (normalizePhoneForLogin(phoneValues[r][0]) === target) {
      var fallbackRowIndex = r + 2;
      var fallbackValues = sheet.getRange(fallbackRowIndex, 1, 1, info.lastColumn).getDisplayValues()[0];
      return { rowIndex: fallbackRowIndex, row: rowToObjectFromValues(headers, fallbackValues) };
    }
  }

  return { rowIndex: -1, row: null };
}

function writeByHeaders(sheet, rowIndex, payloadData) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];

  headers.forEach(function(header, idx) {
    var key = normalizeHeader(header);
    if (payloadData[key] !== undefined) {
      var val = payloadData[key];
      if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
      sheet.getRange(rowIndex, idx + 1).setValue(val);
    }
  });
}

function appendByHeaders(sheet, payloadData) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
  var newRow = [];

  headers.forEach(function(header) {
    var key = normalizeHeader(header);
    var val = payloadData[key] !== undefined ? payloadData[key] : "";
    if (key === 'id' && val === "") val = new Date().getTime().toString();
    if (key === 'pin_diubah' && val === "") val = "Belum";
    if (key === 'pin_wa_terkirim' && val === "") val = "Belum";
    if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
    newRow.push(val);
  });

  sheet.appendRow(newRow);
}

function sortDataRowsByFirstColumn(sheet) {
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort(1);
  }
}


function createAuthSession(role, nohp, nama) {
  var token = Utilities.getUuid() + '_' + new Date().getTime();
  var sessionData = {
    role: role,
    nohp: nohp || '',
    nama: nama || '',
    created_at: nowString()
  };
  CacheService.getScriptCache().put('nexus_auth_' + token, JSON.stringify(sessionData), AUTH_SESSION_TTL_SECONDS);
  return token;
}

function getAuthSession(token) {
  token = String(token || '').trim();
  if (!token) return null;
  var raw = CacheService.getScriptCache().get('nexus_auth_' + token);
  if (!raw) return null;
  try {
    return JSON.parse(raw);
  } catch (err) {
    return null;
  }
}

function requireAuthSession(payload) {
  var token = (payload && (payload.token || (payload.auth && payload.auth.token))) || '';
  var session = getAuthSession(token);
  if (!session || !session.role) throw new Error('Sesi login tidak valid atau sudah berakhir. Silakan login ulang.');
  return session;
}

function requireAdminSession(payload) {
  var session = requireAuthSession(payload);
  if (session.role !== 'admin') throw new Error('Akses ditolak. Fitur ini hanya untuk Admin.');
  return session;
}

function getSecureDataForSession(session) {
  var result = { status: 'success', role: session.role };
  var allPasis = getPublicPasisData();

  result.pasis = allPasis;
  result.pengeluaran = sheetToJSON(getPengeluaranSheet());
  result.pindah_dana = sheetToJSON(getPindahDanaSheet());
  result.pengaturan = sheetToJSON(getPengaturanSheet());
  result.access_views = session.role === 'admin'
    ? ['dashboard', 'pasis', 'akun', 'pengeluaran', 'pindah-dana', 'pengaturan']
    : ['dashboard', 'pasis', 'akun', 'pengeluaran'];

  if (session.role === 'pasis') {
    var targetPhone = normalizePhoneForLogin(session.nohp);
    var rowPasis = allPasis.filter(function(row) {
      return normalizePhoneForLogin(row.nohp) === targetPhone;
    })[0];
    if (!rowPasis) throw new Error('Sesi pasis tidak ditemukan. Silakan login ulang.');

    var pasisCuti = isPasisCuti(rowPasis);
    result.user = rowPasis;
    result.pasis_status = pasisCuti ? 'cuti' : 'aktif';
    result.access_scope = pasisCuti ? 'self' : 'all';

    // Aturan akses:
    // - Pasis Aktif: boleh melihat seluruh Data Pasis dan menu umum.
    // - Pasis Cuti: boleh login, tetapi hanya menerima data dirinya sendiri dan menu Akun Saya.
    if (pasisCuti) {
      result.pasis = [rowPasis];
      result.pengeluaran = [];
      result.pindah_dana = [];
      result.access_views = ['pasis', 'akun'];
    } else {
      result.access_views = ['dashboard', 'pasis', 'akun', 'pengeluaran'];
    }
  }

  return result;
}

function doGet(e) {
  return makeOutput({
    status: 'locked',
    message: 'Data dilindungi. Silakan login melalui aplikasi Nexus.'
  });
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents || '{}');
    if (!payload.action) throw new Error("Action kosong.");

    // --- AMBIL DATA SETELAH LOGIN ---
    if (payload.action === 'get_secure_data') {
      var authSession = requireAuthSession(payload);
      var secureResult = getSecureDataForSession(authSession);
      return makeOutput(secureResult);
    }

    // --- LOGIN ADMIN CEPAT + PROTEKSI 3x SALAH PER PERANGKAT/BROWSER ---
    if (payload.action === 'login_admin') {
      var adminData = payload.data || {};
      var pinAdmin = normalizePinValue(adminData.pin);
      var clientId = adminData.client_id || adminData.device_id || 'global';
      if (!pinAdmin) throw new Error("PIN Admin wajib diisi.");

      var savedAdminPin = getAdminPinValue();

      // PIN benar harus tetap bisa masuk walaupun perangkat sempat terkunci karena salah input.
      // Ini mencegah kasus admin asli tidak bisa login karena lock lama/cache browser.
      if (pinAdmin === savedAdminPin) {
        clearAdminFailedAttempts(clientId);
        var settingsLogin = getSettingsObject();
        return makeOutput({
          status: 'success',
          role: 'admin',
          token: createAuthSession('admin', '', 'Admin'),
          pengaturan: Object.keys(settingsLogin).map(function(key) {
            return { key: key, value: settingsLogin[key] };
          }),
          message: 'Login Admin berhasil.'
        });
      }

      var lockState = getAdminLockState(clientId);
      if (lockState.locked) {
        return makeOutput({
          status: 'locked',
          code: 'ADMIN_LOCKED',
          remaining_seconds: lockState.remaining_seconds,
          remaining_attempts: 0,
          message: 'Login Admin terkunci sementara karena PIN salah 3 kali. Coba lagi dalam ' + Math.ceil(lockState.remaining_seconds / 60) + ' menit.'
        });
      }

      return makeOutput(registerAdminFailedAttempt(clientId));
    }

    // --- LOGIN PASIS ---
    if (payload.action === 'login_pasis') {
      var loginData = payload.data || {};
      var nohpLogin = loginData.nohp;
      var pinLogin = String(loginData.pin || '').trim();
      if (!nohpLogin || !pinLogin) throw new Error("No HP dan PIN wajib diisi.");

      var sheetLogin = getDataPasisSheet();
      var foundLogin = findPasisRowByPhone(sheetLogin, nohpLogin);
      if (foundLogin.rowIndex === -1 || !foundLogin.row) throw new Error("Akun pasis tidak ditemukan.");

      var effectivePin = getEffectivePasisPin(foundLogin.row);
      if (pinLogin !== effectivePin) throw new Error("PIN salah.");

      var loginTime = nowString();
      writeByHeaders(sheetLogin, foundLogin.rowIndex, { last_login: loginTime });
      foundLogin.row.last_login = loginTime;

      var loginIsCuti = isPasisCuti(foundLogin.row);
      return makeOutput({
        status: 'success',
        role: 'pasis',
        token: createAuthSession('pasis', normalizePhoneForLogin(foundLogin.row.nohp), foundLogin.row.nama || 'Pasis'),
        user: sanitizePasisForClient(foundLogin.row),
        pasis_status: loginIsCuti ? 'cuti' : 'aktif',
        access_scope: loginIsCuti ? 'self' : 'all',
        access_views: loginIsCuti ? ['pasis', 'akun'] : ['dashboard', 'pasis', 'akun', 'pengeluaran'],
        default_pin_hint: !foundLogin.row.pin_login ? 'PIN awal adalah 6 digit terakhir No HP.' : '',
        message: loginIsCuti ? 'Login berhasil. Status Cuti: akses hanya Data Pasis milik Anda dan Akun Saya.' : 'Login berhasil.'
      });
    }

    // --- GANTI PIN PASIS ---
    if (payload.action === 'update_pin_pasis') {
      var pinData = payload.data || {};
      var nohpPin = pinData.nohp;
      var oldPin = String(pinData.old_pin || '').trim();
      var newPin = String(pinData.new_pin || '').trim();

      if (!nohpPin || !oldPin || !newPin) throw new Error("No HP, PIN lama, dan PIN baru wajib diisi.");
      if (!/^\d{4,12}$/.test(newPin)) throw new Error("PIN baru harus angka 4 sampai 12 digit.");

      var sheetPin = getDataPasisSheet();
      var foundPin = findPasisRowByPhone(sheetPin, nohpPin);
      if (foundPin.rowIndex === -1 || !foundPin.row) throw new Error("Akun pasis tidak ditemukan.");

      var currentPin = getEffectivePasisPin(foundPin.row);
      if (oldPin !== currentPin) throw new Error("PIN lama salah.");
      if (newPin === currentPin) throw new Error("PIN baru tidak boleh sama dengan PIN lama.");

      writeByHeaders(sheetPin, foundPin.rowIndex, {
        pin_login: newPin,
        pin_diubah: 'Ya',
        last_pin_change: nowString()
      });

      foundPin.row.pin_diubah = 'Ya';
      foundPin.row.last_pin_change = nowString();
      return makeOutput({
        status: 'success',
        message: 'PIN berhasil diperbarui.',
        user: sanitizePasisForClient(foundPin.row)
      });
    }

    // --- SHARE PIN PASIS VIA WHATSAPP (ADMIN) ---
    if (payload.action === 'get_pin_share_list') {
      requireAdminSession(payload);
      return makeOutput({
        status: 'success',
        data: getPinShareDataForAdmin()
      });
    }

    if (payload.action === 'generate_share_pin' || payload.action === 'reset_active_pasis_pins') {
      requireAdminSession(payload);
      var resetList = generatePinsForSharePinPasis();
      return makeOutput({
        status: 'success',
        updated: resetList.length,
        data: resetList,
        message: resetList.length + ' PIN pasis Aktif/Cuti berhasil dibuat ulang.'
      });
    }

    if (payload.action === 'mark_pin_shared') {
      requireAdminSession(payload);
      var shareData = payload.data || {};
      var listPhone = shareData.nohp_list || [];
      if (!Array.isArray(listPhone)) listPhone = [];
      if (shareData.nohp) listPhone.push(shareData.nohp);
      if (listPhone.length === 0) throw new Error("Daftar No HP kosong.");

      var sheetShare = getDataPasisSheet();
      var updatedCount = 0;
      listPhone.forEach(function(phone) {
        if (markPinSharedByPhone(sheetShare, phone)) updatedCount++;
      });

      return makeOutput({
        status: 'success',
        updated: updatedCount,
        message: updatedCount + ' status share PIN diperbarui.'
      });
    }

    // --- AKSI PING (KUNJUNGAN & USER AKTIF) ---
    if (payload.action === 'ping') {
      var cache = CacheService.getScriptCache();
      var activeUsersStr = cache.get('nexus_active_users');
      var activeUsers = activeUsersStr ? JSON.parse(activeUsersStr) : {};
      var now = new Date().getTime();

      if (payload.sessionId) activeUsers[payload.sessionId] = now;

      var activeCount = 0;
      for (var key in activeUsers) {
        if (now - activeUsers[key] > 120000) delete activeUsers[key];
        else activeCount++;
      }
      cache.put('nexus_active_users', JSON.stringify(activeUsers), 300);

      var sheetPengaturan = getPengaturanSheet();
      var dataPengaturan = sheetPengaturan.getDataRange().getDisplayValues();
      var totalViews = 0;
      var viewRowIndex = -1;

      for (var r = 1; r < dataPengaturan.length; r++) {
        if (normalizeHeader(dataPengaturan[r][0]) === 'total_views') {
          viewRowIndex = r + 1;
          totalViews = parseInt(dataPengaturan[r][1]) || 0;
          break;
        }
      }

      if (payload.isNewVisit) {
        totalViews++;
        if (viewRowIndex > -1) sheetPengaturan.getRange(viewRowIndex, 2).setValue(totalViews);
        else sheetPengaturan.appendRow(['total_views', totalViews]);
      }

      return makeOutput({ status: 'success', activeUsers: activeCount, totalViews: totalViews });
    }

    // --- AKSI DATA PASIS ---
    if (payload.action === 'create' || payload.action === 'update') {
      requireAdminSession(payload);
      var sheetPasis = getDataPasisSheet();

      if (payload.action === 'create') {
        var createData = payload.data || {};
        if (!createData.pin_diubah) createData.pin_diubah = 'Belum';
        appendByHeaders(sheetPasis, createData);
        sortDataRowsByFirstColumn(sheetPasis);
        return makeOutput({ status: 'success' });
      }

      if (payload.action === 'update') {
        var data = payload.data || {};
        var targetNohp = data.original_nohp || data.nohp;
        if (!targetNohp) throw new Error("Nomor HP target kosong.");

        var rowIndex = findRowByHeaderValue(sheetPasis, 'nohp', targetNohp);
        if (rowIndex === -1) throw new Error("Data tidak ditemukan.");

        writeByHeaders(sheetPasis, rowIndex, data);
        return makeOutput({ status: 'success' });
      }
    }

    if (payload.action === 'delete_pasis') {
      requireAdminSession(payload);
      var sheetDeletePasis = getDataPasisSheet();
      var rowDelete = findRowByHeaderValue(sheetDeletePasis, 'nohp', payload.data && payload.data.nohp);
      if (rowDelete === -1) throw new Error("Pasis tidak ditemukan.");

      sheetDeletePasis.deleteRow(rowDelete);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI PENGELUARAN ---
    if (payload.action === 'create_pengeluaran') {
      requireAdminSession(payload);
      appendByHeaders(getPengeluaranSheet(), payload.data || {});
      return makeOutput({ status: 'success' });
    }

    if (payload.action === 'delete_pengeluaran') {
      requireAdminSession(payload);
      var sheetPengeluaran = getPengeluaranSheet();
      var rowPengeluaran = findRowByHeaderValue(sheetPengeluaran, 'id', payload.data && payload.data.id);
      if (rowPengeluaran === -1) throw new Error("Pengeluaran tidak ditemukan.");

      sheetPengeluaran.deleteRow(rowPengeluaran);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI PINDAH DANA / SERAH TERIMA KE KETUA KELAS ---
    if (payload.action === 'create_pindah_dana') {
      requireAdminSession(payload);
      appendByHeaders(getPindahDanaSheet(), payload.data || {});
      return makeOutput({ status: 'success' });
    }

    if (payload.action === 'delete_pindah_dana') {
      requireAdminSession(payload);
      var sheetPindah = getPindahDanaSheet();
      var rowPindah = findRowByHeaderValue(sheetPindah, 'id', payload.data && payload.data.id);
      if (rowPindah === -1) throw new Error("Data serah terima kas tidak ditemukan.");

      sheetPindah.deleteRow(rowPindah);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI UPDATE PENGATURAN ---
    if (payload.action === 'update_pengaturan') {
      requireAdminSession(payload);
      var sheetSetting = getPengaturanSheet();
      var settingData = sheetSetting.getDataRange().getDisplayValues();
      var keysToUpdate = payload.data || {};

      for (var k in keysToUpdate) {
        var found = false;
        for (var s = 1; s < settingData.length; s++) {
          if (normalizeHeader(settingData[s][0]) === normalizeHeader(k)) {
            sheetSetting.getRange(s + 1, 2).setValue(keysToUpdate[k]);
            found = true;
            break;
          }
        }
        if (!found) sheetSetting.appendRow([k, keysToUpdate[k]]);
      }

      return makeOutput({ status: 'success' });
    }

    throw new Error("Action tidak valid: " + payload.action);
  } catch (err) {
    return makeOutput({ status: 'error', message: err.toString() });
  }
}
