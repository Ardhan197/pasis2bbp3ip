/**
 * API BACKEND NEXUS
 * PASIS + PENGELUARAN + PINDAH DANA + PENGATURAN + VISITOR TRAFFIC
 * Update: target cuti/refund global + indikator refund sukses + dashboard premium + share pengeluaran WhatsApp.
 */

var DATA_PASIS_HEADERS = [
  'nama', 'nohp', 'jk', 'bank', 'norek',
  'dibayar', 'tgl_bayar', 'riwayat', 'status_aktif',
  'target_khusus', 'refund', 'tgl_refund', 'catatan_refund',
  'status_refund', 'waktu_refund'
];

var PENGELUARAN_HEADERS = ['id', 'tanggal', 'keterangan', 'nominal'];
var PINDAH_DANA_HEADERS = ['id', 'tanggal', 'penerima', 'metode', 'keterangan', 'nominal'];
var PENGATURAN_HEADERS = ['key', 'value'];

var DEFAULT_SETTINGS = {
  target_tagihan: 15000000,
  target_cuti_refund: 0
};

function getSpreadsheetSafe() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error("Koneksi ditolak. Pastikan script ditempel pada Apps Script milik Google Sheets.");
  return ss;
}

function normalizeHeader(value) {
  return String(value || '').toLowerCase().trim();
}

function ensureHeaders(sheet, requiredHeaders) {
  if (!sheet) throw new Error("Sheet tidak valid.");

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(requiredHeaders);
    return sheet;
  }

  var lastColumn = Math.max(sheet.getLastColumn(), 1);
  var headerValues = sheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  var existing = {};
  headerValues.forEach(function(header) {
    existing[normalizeHeader(header)] = true;
  });

  requiredHeaders.forEach(function(header) {
    if (!existing[normalizeHeader(header)]) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(header);
      existing[normalizeHeader(header)] = true;
    }
  });

  return sheet;
}

function getOrCreateSheet(name, headers) {
  var ss = getSpreadsheetSafe();
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return ensureHeaders(sheet, headers);
}

function getSheetSafe(name) {
  var ss = getSpreadsheetSafe();
  var sheet = ss.getSheetByName(name);
  if (!sheet) throw new Error("Tab '" + name + "' tidak ditemukan.");
  return sheet;
}

function getDataPasisSheet() {
  return getOrCreateSheet("Data Pasis", DATA_PASIS_HEADERS);
}

function getPengeluaranSheet() {
  return getOrCreateSheet("Pengeluaran", PENGELUARAN_HEADERS);
}

function getPindahDanaSheet() {
  return getOrCreateSheet("Pindah Dana", PINDAH_DANA_HEADERS);
}

function getPengaturanSheet() {
  var sheet = getOrCreateSheet("Pengaturan", PENGATURAN_HEADERS);
  ensureDefaultSettings(sheet);
  return sheet;
}

function ensureDefaultSettings(sheet) {
  var data = sheet.getDataRange().getDisplayValues();
  var existing = {};
  for (var r = 1; r < data.length; r++) {
    existing[normalizeHeader(data[r][0])] = true;
  }

  for (var key in DEFAULT_SETTINGS) {
    if (!existing[normalizeHeader(key)]) {
      sheet.appendRow([key, DEFAULT_SETTINGS[key]]);
      existing[normalizeHeader(key)] = true;
    }
  }
}

function sheetToJSON(sheet) {
  if (!sheet) return [];
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return [];

  var headers = data[0].map(function(header) {
    return normalizeHeader(header);
  });

  return data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(header, index) {
      if (header) obj[header] = row[index];
    });
    return obj;
  }).filter(function(r) {
    var firstKey = headers[0];
    return r[firstKey] && r[firstKey] !== "";
  });
}

function makeOutput(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function findRowByHeaderValue(sheet, headerName, targetValue) {
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return -1;

  var headers = data[0].map(normalizeHeader);
  var targetIndex = headers.indexOf(normalizeHeader(headerName));
  if (targetIndex === -1) throw new Error("Kolom '" + headerName + "' tidak ditemukan.");

  var target = String(targetValue || '').replace(/^'/, '').trim();
  for (var r = 1; r < data.length; r++) {
    var cell = String(data[r][targetIndex] || '').replace(/^'/, '').trim();
    if (cell === target) return r + 1;
  }

  return -1;
}

function writeByHeaders(sheet, rowIndex, payloadData) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];

  headers.forEach(function(header, idx) {
    var key = normalizeHeader(header);
    if (payloadData[key] !== undefined) {
      var val = payloadData[key];
      if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
      sheet.getRange(rowIndex, idx + 1).setValue(val);
    }
  });
}

function appendByHeaders(sheet, payloadData) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
  var newRow = [];

  headers.forEach(function(header) {
    var key = normalizeHeader(header);
    var val = payloadData[key] !== undefined ? payloadData[key] : "";
    if (key === 'id' && val === "") val = new Date().getTime().toString();
    if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
    newRow.push(val);
  });

  sheet.appendRow(newRow);
}

function sortDataRowsByFirstColumn(sheet) {
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort(1);
  }
}

function doGet(e) {
  try {
    var result = { status: 'success' };

    try { result.pasis = sheetToJSON(getDataPasisSheet()); } catch (errPasis) { result.pasis = []; }
    try { result.pengeluaran = sheetToJSON(getPengeluaranSheet()); } catch (errPengeluaran) { result.pengeluaran = []; }
    try { result.pindah_dana = sheetToJSON(getPindahDanaSheet()); } catch (errPindah) { result.pindah_dana = []; }
    try { result.pengaturan = sheetToJSON(getPengaturanSheet()); } catch (errPengaturan) { result.pengaturan = []; }

    return makeOutput(result);
  } catch (error) {
    return makeOutput({ status: 'error', message: error.toString() });
  }
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents || '{}');
    if (!payload.action) throw new Error("Action kosong.");

    // --- AKSI PING (KUNJUNGAN & USER AKTIF) ---
    if (payload.action === 'ping') {
      var cache = CacheService.getScriptCache();
      var activeUsersStr = cache.get('nexus_active_users');
      var activeUsers = activeUsersStr ? JSON.parse(activeUsersStr) : {};
      var now = new Date().getTime();

      if (payload.sessionId) activeUsers[payload.sessionId] = now;

      var activeCount = 0;
      for (var key in activeUsers) {
        if (now - activeUsers[key] > 120000) delete activeUsers[key];
        else activeCount++;
      }
      cache.put('nexus_active_users', JSON.stringify(activeUsers), 300);

      var sheetPengaturan = getPengaturanSheet();
      var dataPengaturan = sheetPengaturan.getDataRange().getDisplayValues();
      var totalViews = 0;
      var viewRowIndex = -1;

      for (var r = 1; r < dataPengaturan.length; r++) {
        if (normalizeHeader(dataPengaturan[r][0]) === 'total_views') {
          viewRowIndex = r + 1;
          totalViews = parseInt(dataPengaturan[r][1]) || 0;
          break;
        }
      }

      if (payload.isNewVisit) {
        totalViews++;
        if (viewRowIndex > -1) sheetPengaturan.getRange(viewRowIndex, 2).setValue(totalViews);
        else sheetPengaturan.appendRow(['total_views', totalViews]);
      }

      return makeOutput({ status: 'success', activeUsers: activeCount, totalViews: totalViews });
    }

    // --- AKSI DATA PASIS ---
    if (payload.action === 'create' || payload.action === 'update') {
      var sheetPasis = getDataPasisSheet();

      if (payload.action === 'create') {
        appendByHeaders(sheetPasis, payload.data || {});
        sortDataRowsByFirstColumn(sheetPasis);
        return makeOutput({ status: 'success' });
      }

      if (payload.action === 'update') {
        var data = payload.data || {};
        var targetNohp = data.original_nohp || data.nohp;
        if (!targetNohp) throw new Error("Nomor HP target kosong.");

        var rowIndex = findRowByHeaderValue(sheetPasis, 'nohp', targetNohp);
        if (rowIndex === -1) throw new Error("Data tidak ditemukan.");

        writeByHeaders(sheetPasis, rowIndex, data);
        return makeOutput({ status: 'success' });
      }
    }

    if (payload.action === 'delete_pasis') {
      var sheetDeletePasis = getDataPasisSheet();
      var rowDelete = findRowByHeaderValue(sheetDeletePasis, 'nohp', payload.data && payload.data.nohp);
      if (rowDelete === -1) throw new Error("Pasis tidak ditemukan.");

      sheetDeletePasis.deleteRow(rowDelete);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI PENGELUARAN ---
    if (payload.action === 'create_pengeluaran') {
      appendByHeaders(getPengeluaranSheet(), payload.data || {});
      return makeOutput({ status: 'success' });
    }

    if (payload.action === 'delete_pengeluaran') {
      var sheetPengeluaran = getPengeluaranSheet();
      var rowPengeluaran = findRowByHeaderValue(sheetPengeluaran, 'id', payload.data && payload.data.id);
      if (rowPengeluaran === -1) throw new Error("Pengeluaran tidak ditemukan.");

      sheetPengeluaran.deleteRow(rowPengeluaran);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI PINDAH DANA / SERAH TERIMA KE KETUA KELAS ---
    if (payload.action === 'create_pindah_dana') {
      appendByHeaders(getPindahDanaSheet(), payload.data || {});
      return makeOutput({ status: 'success' });
    }

    if (payload.action === 'delete_pindah_dana') {
      var sheetPindah = getPindahDanaSheet();
      var rowPindah = findRowByHeaderValue(sheetPindah, 'id', payload.data && payload.data.id);
      if (rowPindah === -1) throw new Error("Data serah terima kas tidak ditemukan.");

      sheetPindah.deleteRow(rowPindah);
      return makeOutput({ status: 'success' });
    }

    // --- AKSI UPDATE PENGATURAN ---
    if (payload.action === 'update_pengaturan') {
      var sheetSetting = getPengaturanSheet();
      var settingData = sheetSetting.getDataRange().getDisplayValues();
      var keysToUpdate = payload.data || {};

      for (var k in keysToUpdate) {
        var found = false;
        for (var s = 1; s < settingData.length; s++) {
          if (normalizeHeader(settingData[s][0]) === normalizeHeader(k)) {
            sheetSetting.getRange(s + 1, 2).setValue(keysToUpdate[k]);
            found = true;
            break;
          }
        }
        if (!found) sheetSetting.appendRow([k, keysToUpdate[k]]);
      }

      return makeOutput({ status: 'success' });
    }

    throw new Error("Action tidak valid: " + payload.action);
  } catch (err) {
    return makeOutput({ status: 'error', message: err.toString() });
  }
}
