/**
 * API BACKEND NEXUS (PASIS + PENGELUARAN + PINDAH DANA + PENGATURAN + VISITOR TRAFFIC)
 */

function getSheetSafe(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error("Koneksi ditolak.");
  var sheet = ss.getSheetByName(name);
  if (!sheet) throw new Error("Tab '" + name + "' tidak ditemukan.");
  return sheet;
}

function getOrCreateSheet(name, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error("Koneksi ditolak.");
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  if (sheet.getLastRow() === 0) sheet.appendRow(headers);
  return sheet;
}

function sheetToJSON(sheet) {
  if(!sheet) return [];
  var data = sheet.getDataRange().getDisplayValues(); 
  if(data.length < 2) return [];
  var headers = data[0];
  return data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(header, index) { obj[header.toString().toLowerCase().trim()] = row[index]; });
    return obj;
  }).filter(function(r) { 
     var firstKey = headers[0].toString().toLowerCase().trim();
     return r[firstKey] && r[firstKey] !== ""; 
  });
}

function doGet(e) {
  try {
    var result = { status: 'success' };
    try { result.pasis = sheetToJSON(getSheetSafe("Data Pasis")); } catch(e) { result.pasis = []; }
    try { result.pengeluaran = sheetToJSON(getSheetSafe("Pengeluaran")); } catch(e) { result.pengeluaran = []; }
    try { result.pindah_dana = sheetToJSON(getSheetSafe("Pindah Dana")); } catch(e) { result.pindah_dana = []; }
    try { result.pengaturan = sheetToJSON(getSheetSafe("Pengaturan")); } catch(e) { result.pengaturan = []; }
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
     return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    
    // --- AKSI PING (KUNJUNGAN & USER AKTIF) ---
    if (payload.action === 'ping') {
        var cache = CacheService.getScriptCache();
        var activeUsersStr = cache.get('nexus_active_users');
        var activeUsers = activeUsersStr ? JSON.parse(activeUsersStr) : {};
        var now = new Date().getTime();

        if (payload.sessionId) activeUsers[payload.sessionId] = now;

        var activeCount = 0;
        for (var key in activeUsers) {
            if (now - activeUsers[key] > 120000) { delete activeUsers[key]; } // Expire 2 min
            else { activeCount++; }
        }
        cache.put('nexus_active_users', JSON.stringify(activeUsers), 300);

        var sheet = getOrCreateSheet("Pengaturan", ['key', 'value']);
        var totalViews = 0;
        if (sheet) {
            var data = sheet.getDataRange().getDisplayValues();
            var viewRowIndex = -1;
            for (var r = 1; r < data.length; r++) {
                if (data[r][0].toString().toLowerCase().trim() === 'total_views') {
                    viewRowIndex = r + 1;
                    totalViews = parseInt(data[r][1]) || 0;
                    break;
                }
            }
            if (payload.isNewVisit) {
                totalViews++;
                if (viewRowIndex > -1) sheet.getRange(viewRowIndex, 2).setValue(totalViews);
                else sheet.appendRow(['total_views', totalViews]);
            }
        }

        return ContentService.createTextOutput(JSON.stringify({ status: 'success', activeUsers: activeCount, totalViews: totalViews })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // --- AKSI DATA PASIS ---
    if (payload.action === 'create' || payload.action === 'update') {
      var sheet = getSheetSafe("Data Pasis");
      if(!sheet) throw new Error("Sheet 'Data Pasis' tidak ditemukan.");
      var data = sheet.getDataRange().getDisplayValues();
      var headers = data[0];
      
      if (payload.action === 'create') {
        var newRow = [];
        headers.forEach(function(header) {
          var key = header.toString().toLowerCase().trim();
          var val = payload.data[key] !== undefined ? payload.data[key] : "";
          if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
          newRow.push(val);
        });
        sheet.appendRow(newRow);
        
        // OTOMATIS MENGURUTKAN DATABASE GOOGLE SHEETS (A-Z) BERDASARKAN KOLOM A (NAMA)
        if (sheet.getLastRow() > 1) {
          sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).sort(1);
        }

        return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
      }
      
      if (payload.action === 'update') {
        var nohpIndex = -1;
        for(var i = 0; i < headers.length; i++) { if(headers[i].toString().toLowerCase().trim() === 'nohp') { nohpIndex = i; break; } }
        if(nohpIndex === -1) throw new Error("Kolom 'nohp' tidak ditemukan.");
        
        var targetNohp = payload.data.original_nohp || payload.data.nohp;
        var rowIndex = -1;
        for(var r = 1; r < data.length; r++) { if(data[r][nohpIndex] == targetNohp) { rowIndex = r + 1; break; } }
        
        if (rowIndex > -1) {
          headers.forEach(function(header, idx) {
            var key = header.toString().toLowerCase().trim();
            if (payload.data[key] !== undefined) {
              var val = payload.data[key];
              if (val !== "" && (key === 'nohp' || key === 'norek')) val = "'" + val;
              sheet.getRange(rowIndex, idx + 1).setValue(val);
            }
          });
          return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
        } else { throw new Error("Data tidak ditemukan."); }
      }
    }
    
    if (payload.action === 'delete_pasis') {
      var sheet = getSheetSafe("Data Pasis");
      var data = sheet.getDataRange().getDisplayValues();
      var nohpIndex = -1;
      for(var i = 0; i < data[0].length; i++) { if(data[0][i].toString().toLowerCase().trim() === 'nohp') { nohpIndex = i; break; } }
      if(nohpIndex === -1) throw new Error("Kolom 'nohp' tidak ditemukan.");
      
      var rowIndex = -1;
      for(var r = 1; r < data.length; r++) { if(data[r][nohpIndex] == payload.data.nohp) { rowIndex = r + 1; break; } }
      if (rowIndex > -1) {
        sheet.deleteRow(rowIndex);
        return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
      } else { throw new Error("Pasis tidak ditemukan."); }
    }

    // --- AKSI PENGELUARAN ---
    if (payload.action === 'create_pengeluaran') {
       var sheet = getOrCreateSheet("Pengeluaran", ['id', 'tanggal', 'keterangan', 'nominal']);
       var headers = sheet.getDataRange().getDisplayValues()[0];
       var newRow = [];
       headers.forEach(function(header) {
         var key = header.toString().toLowerCase().trim();
         var val = payload.data[key] !== undefined ? payload.data[key] : "";
         if (key === 'id' && val === "") val = new Date().getTime().toString();
         newRow.push(val);
       });
       sheet.appendRow(newRow);
       return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === 'delete_pengeluaran') {
       var sheet = getSheetSafe("Pengeluaran");
       var data = sheet.getDataRange().getDisplayValues();
       var idIndex = -1;
       for(var i = 0; i < data[0].length; i++) { if(data[0][i].toString().toLowerCase().trim() === 'id') { idIndex = i; break; } }
       var rowIndex = -1;
       for(var r = 1; r < data.length; r++) { if(data[r][idIndex] == payload.data.id) { rowIndex = r + 1; break; } }
       if(rowIndex > -1) { sheet.deleteRow(rowIndex); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON); } 
       else { throw new Error("Pengeluaran tidak ditemukan."); }
    }

    // --- AKSI PINDAH DANA / SERAH TERIMA KE KETUA KELAS ---
    if (payload.action === 'create_pindah_dana') {
       var sheet = getOrCreateSheet("Pindah Dana", ['id', 'tanggal', 'penerima', 'metode', 'keterangan', 'nominal']);
       var headers = sheet.getDataRange().getDisplayValues()[0];
       var newRow = [];
       headers.forEach(function(header) {
         var key = header.toString().toLowerCase().trim();
         var val = payload.data[key] !== undefined ? payload.data[key] : "";
         if (key === 'id' && val === "") val = new Date().getTime().toString();
         newRow.push(val);
       });
       sheet.appendRow(newRow);
       return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === 'delete_pindah_dana') {
       var sheet = getSheetSafe("Pindah Dana");
       var data = sheet.getDataRange().getDisplayValues();
       var idIndex = -1;
       for(var i = 0; i < data[0].length; i++) { if(data[0][i].toString().toLowerCase().trim() === 'id') { idIndex = i; break; } }
       if(idIndex === -1) throw new Error("Kolom 'id' tidak ditemukan.");
       var rowIndex = -1;
       for(var r = 1; r < data.length; r++) { if(data[r][idIndex] == payload.data.id) { rowIndex = r + 1; break; } }
       if(rowIndex > -1) { sheet.deleteRow(rowIndex); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON); } 
       else { throw new Error("Data pindah dana tidak ditemukan."); }
    }

    // --- AKSI UPDATE PENGATURAN ---
    if (payload.action === 'update_pengaturan') {
        var sheet = getOrCreateSheet("Pengaturan", ['key', 'value']);
        var data = sheet.getDataRange().getDisplayValues();
        var keysToUpdate = payload.data; 
        for (var k in keysToUpdate) {
            var found = false;
            for (var r = 1; r < data.length; r++) {
                if (data[r][0].toString().toLowerCase().trim() === k.toLowerCase()) {
                    sheet.getRange(r + 1, 2).setValue(keysToUpdate[k]); found = true; break;
                }
            }
            if (!found) sheet.appendRow([k, keysToUpdate[k]]);
        }
        return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
    }

    throw new Error("Action tidak valid.");
  } catch (err) {
     return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

Struktur Database

1

Sheet: Data Pasis

Kolom A-I: nama, nohp, jk, bank, norek, dibayar, tgl_bayar, riwayat, status_aktif

2

Sheet: Pengeluaran

Kolom A-D: id, tanggal, keterangan, nominal

3

Sheet: Pindah Dana

Kolom A-F: id, tanggal, penerima, metode, keterangan, nominal. Sheet ini mencatat dana yang diserahkan ke ketua kelas.

4

Sheet: Pengaturan

Kolom A & B: key, value. (Sistem akan membuat total_views otomatis)
