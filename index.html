Kode Google Apps Script

PANDUAN PENTING:

  1. Buka Editor Apps Script, HAPUS SEMUA kode lama.
  2. Tempel kode baru dari tombol di atas. Klik Save.
  3. Klik Deploy > New Deployment.
  4. "Who has access": Anyone. (Pilih ini agar tidak Error!)
  5. Klik Deploy dan berikan izin.
function doPost(e) {
  try {
    var data;
    if (e.postData.type == "application/json") {
       data = JSON.parse(e.postData.contents);
    } else {
       data = JSON.parse(e.postData.contents);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Sinkronisasi Data Pasis
    var sheetPasis = ss.getSheetByName("Data Pasis") || ss.insertSheet("Data Pasis");
    sheetPasis.clear();
    var headersPasis = ["ID", "Nama Lengkap", "NRP/Pangkat", "Bank", "No Rekening", "No HP"];
    sheetPasis.appendRow(headersPasis);
    sheetPasis.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#D1E9FF");
    
    if (data.pasis && data.pasis.length > 0) {
      data.pasis.forEach(function(p) {
        sheetPasis.appendRow(["'" + p.id, p.name, "'" + p.nrp, p.bank || "-", "'" + (p.rek || "-"), "'" + (p.hp || "-")]);
      });
    }

    // 2. Sinkronisasi Data Transaksi
    var sheetTrx = ss.getSheetByName("Data Transaksi") || ss.insertSheet("Data Transaksi");
    sheetTrx.clear();
    var headersTrx = ["ID", "Tanggal", "Deskripsi", "Tipe", "Nominal", "Penyetor"];
    sheetTrx.appendRow(headersTrx);
    sheetTrx.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#D1FAE5");
    
    if (data.transactions && data.transactions.length > 0) {
      data.transactions.forEach(function(t) {
        sheetTrx.appendRow(["'" + t.id, t.date, t.description, t.type, t.amount, t.pasisName || "-"]);
      });
    }

    sheetPasis.autoResizeColumns(1, 6);
    sheetTrx.autoResizeColumns(1, 6);

    return ContentService.createTextOutput("Sinkronisasi Berhasil").setMimeType(ContentService.MimeType.TEXT);
  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.toString()).setMimeType(ContentService.MimeType.TEXT);
  }
}
